	private inFlight: Promise<void> | null = null;
	private loadEpoch = 0;

	subscribe(listener: (state: LocalModel) => void): () => void {
		this.listeners.add(listener);
		listener(this.state);
		return () => this.listeners.delete(listener);
	}

	private notify(): void {
		this.listeners.forEach((l) => l(this.state));
