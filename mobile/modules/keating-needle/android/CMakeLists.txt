cmake_minimum_required(VERSION 3.22.1)
project(keating_needle LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(ASSETS "${CMAKE_CURRENT_LIST_DIR}/../.generated/b274efcb211a9eef48c9a88da4b43bd569696a39")
add_library(keating_needle SHARED needle-jni.cpp)
# Load on both 4 KB and 16 KB Android page-size devices.
target_link_options(keating_needle PRIVATE "-Wl,-z,max-page-size=16384")
target_include_directories(keating_needle PRIVATE "${ASSETS}")
if(ANDROID_ABI STREQUAL "arm64-v8a")
  add_library(needle_pinned STATIC IMPORTED)
  set_target_properties(needle_pinned PROPERTIES IMPORTED_LOCATION "${ASSETS}/android-arm64/libneedle.a")
  target_include_directories(keating_needle PRIVATE "${ASSETS}/android-arm64")
  target_link_libraries(keating_needle PRIVATE needle_pinned log android m)
endif()
