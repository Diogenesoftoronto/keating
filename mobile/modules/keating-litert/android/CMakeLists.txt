cmake_minimum_required(VERSION 3.22.1)
project(keating_label_scorer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(SDK "${CMAKE_CURRENT_LIST_DIR}/../.generated/0.16.0")
add_library(keating_label_scorer SHARED label-scorer-jni.cpp ../cpp/label-scorer.cpp)
target_link_options(keating_label_scorer PRIVATE "-Wl,-z,max-page-size=16384")
target_include_directories(keating_label_scorer PRIVATE "${SDK}/include")
if(ANDROID_ABI STREQUAL "arm64-v8a")
  set(SDK_ABI android_arm64)
elseif(ANDROID_ABI STREQUAL "x86_64")
  set(SDK_ABI android_x86_64)
else()
  message(FATAL_ERROR "Offline judgement requires arm64-v8a or x86_64")
endif()
add_library(litert_scoring SHARED IMPORTED)
set_target_properties(litert_scoring PROPERTIES IMPORTED_LOCATION "${SDK}/lib/${SDK_ABI}/liblitert-lm.so")
target_link_libraries(keating_label_scorer PRIVATE litert_scoring android log)
