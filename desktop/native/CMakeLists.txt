cmake_minimum_required(VERSION 3.20)
project(keating_offline C)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
add_executable(keating-offline runner.c)
target_include_directories(keating-offline PRIVATE "${LITERT_SDK}/include")
target_link_libraries(keating-offline PRIVATE "${LITERT_LIBRARY}")
set_target_properties(keating-offline PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN;@loader_path")
install(TARGETS keating-offline RUNTIME DESTINATION .)
